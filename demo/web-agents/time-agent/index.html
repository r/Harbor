A simple demo that calls the time.now MCP tool.

MCP Tool Call

Click the button to call the time.now tool via the Web Agent API.

Code

// First, ask permission for this specific tool.
await window.agent.requestPermissions({
  scopes: ['mcp:tools.call'],
  tools: ['time-wasm/time.now']
});
const result = await window.agent.tools.call({
  tool: 'time-wasm/time.now',
  args: {}
});
console.log(result);
